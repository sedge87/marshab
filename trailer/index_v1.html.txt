    <script>
      const tl = gsap.timeline({ paused: true });
      // Black dips: open from black, into the touchdown, after the touchdown
      // settles, into night, and out at the end.
      const dip = (at, down, up) => {
        tl.to("#fade", { opacity: 1, duration: down, ease: "power1.in" }, at - down);
        tl.to("#fade", { opacity: 0, duration: up, ease: "power1.out" }, at);
      };
      tl.fromTo("#fade", { opacity: 1 }, { opacity: 0, duration: 2.4, ease: "power1.inOut" }, 0);
      dip(6.5, 0.6, 0.9);
      dip(14.5, 0.35, 0.6);
      dip(40, 0.5, 0.8);
      dip(53, 0.6, 1.2);
      tl.to("#fade", { opacity: 0.55, duration: 1.6, ease: "power1.inOut" }, 54.2);   // dim the finale under the title
      tl.to("#fade", { opacity: 1, duration: 1.4, ease: "power1.in" }, 60.4);

      // Lines: slow rise and fade in, drift, fade out.
      const lines = [["#l1", 1.2, 4.4], ["#l2", 8, 4.6], ["#l3", 20.2, 4.4], ["#l4", 26.2, 4.2], ["#l5", 31.3, 3.9], ["#l6", 35.9, 3.7], ["#l7", 40.6, 4.4], ["#l8", 46, 6.4]];
      for (const [sel, start, dur] of lines) {
        const p = sel + " p";
        tl.fromTo(p, { opacity: 0, y: 14 }, { opacity: 1, y: 0, duration: 1.1, ease: "power2.out" }, start);
        tl.to(p, { opacity: 0, y: -6, duration: 0.8, ease: "power1.in" }, start + dur - 0.8);
      }

      // Title card.
      tl.fromTo("#eyebrow", { opacity: 0, y: 10 }, { opacity: 1, y: 0, duration: 1.0, ease: "power2.out" }, 54.8);
